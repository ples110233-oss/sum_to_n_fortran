一 & 三、 Fortran 原始碼（含逐行中文註解）

!====================================================
! 程式名稱: sum_to_n.f90
! 功能描述: 讓使用者輸入整數 N，計算從 1 加到 N 的總和
!====================================================
program sum_to_n
    implicit none
    integer :: n, i, total_sum  ! 宣告變數：n為輸入上限，i為計數器，total_sum為總和

    ! 提示使用者輸入
    print *, "請輸入一個正整數 N:"
    read *, n

    ! 初始化總和為 0
    total_sum = 0

    ! 使用迴圈從 1 累加到 N
    do i = 1, n
        total_sum = total_sum + i
    end do

    ! 輸出最終計算結果
    print *, "從 1 加到 ", n, " 的總和為: ", total_sum

end program sum_to_n
